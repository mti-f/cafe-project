// ========================= // انتخاب المنت‌ها // ========================= const orderIdInput = document.querySelector("#orderId"); const searchBtn = document.querySelector("#searchOrder"); const orderResult = document.querySelector("#orderResult"); // ========================= // جستجوی سفارش // ========================= searchBtn.addEventListener( "click", function(){ const id = Number(orderIdInput.value.trim()); if(!id){ orderResult.textContent = "لطفاً شماره سفارش را وارد کنید"; return; } const orders = JSON.parse( localStorage.getItem("orders") ) || []; const order = orders.find(function(item){ return item.id === id; }); if(!order){ orderResult.innerHTML = `

❌ سفارشی با این شماره پیدا نشد

`; return; } orderResult.innerHTML = `

سفارش شماره: ${order.id}

نام مشتری: ${order.customer}

مبلغ: ${order.total} تومان

وضعیت سفارش: ${order.status || "در انتظار آماده سازی"}

محصولات:

${ order.items.map(function(item){ return `

${item.name} × ${item.quantity}

` }).join("") }
`; });
